on Scroll-BookPage($ctrl,[int]$delta) {
    try {
        $p=$ctrl
        while($null -ne $p){
            if($p.AutoScroll){
                $old=[math]::Abs([int]$p.AutoScrollPosition.Y)
                $target=$old-[int]($delta/120)*70
                $max=[math]::Max(0,$p.VerticalScroll.Maximum-$p.VerticalScroll.LargeChange+1)
                $target=[math]::Max(0,[math]::Min($target,$max))
                $p.AutoScrollPosition=New-Object System.Drawing.Point(0,$target)
                return
            }
            $p=$p.Parent
        }
    }catch{}
}

    $src=$src.Replace('function New-PageContent([System.Windows.Forms.Panel]$page) {',$scrollFn+'function New-PageContent([System.Windows.Forms.Panel]$page) {')
