Section([string]$Target) {
    if([string]::IsNullOrWhiteSpace($Target) -or $Target -eq $script:CurrentSection){return}
    $from=[array]::IndexOf($script:SectionOrder,$script:CurrentSection);$to=[array]::IndexOf($script:SectionOrder,$Target)
    $direction=1;$distance=1
    if($from -ge 0 -and $to -ge 0){$direction=if($to -gt $from){1}else{-1};$distance=[math]::Max(1,[math]::Abs($to-$from))}
    Animate-BookPages $direction $distance $false
    $script:CurrentSection=$Target
    if($null -ne $script:Settings){$script:Settings.LastSection=$Target}
    Render-CurrentCharacter;Schedule-Save
}
